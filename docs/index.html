export function CheckoutForm() {
  const [state, action, pending] =
    useActionState(placeOrder, initialState);

  return (
    <form action={action}>
      <CartFields />
      {state.error && <Alert>{state.error}</Alert>}
      <SubmitButton disabled={pending} />
    </form>
  );
}
